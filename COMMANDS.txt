-----------------------------------
CHECK REVENUE BY PRODUCT
-----------------------------------

SELECT 
    p.product_name,
    SUM(s.quantity) AS total_units_sold,
    SUM(s.total_amount) AS total_revenue
FROM sales s
JOIN products p 
    ON s.product_id = p.product_id
GROUP BY p.product_name
ORDER BY total_revenue DESC;
